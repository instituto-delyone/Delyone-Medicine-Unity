let focusModeActive = false; function clearHighlights() { document.querySelectorAll('.mapped-field').forEach(el => el.classList.remove('mapped-field')); const btn = $('focusModeBtn'); if (btn) btn.style.display = 'none'; // Esconde o botão if (focusModeActive) toggleFocusMode(); // Desliga o modo foco se a área for limpa } function highlightMappedFields(fileName) { clearHighlights(); let mapToUse = null; if (fileName.includes('AIH')) mapToUse = STATIC_MAPS.AIH; else if (fileName.includes('APAC')) mapToUse = STATIC_MAPS.APAC; if (mapToUse) { for (const key in mapToUse) { const inputEl = $('master_' + mapToUse[key].nativeKey); if (inputEl) { inputEl.classList.add('mapped-field'); const toggleContent = inputEl.closest('.toggle-content'); if (toggleContent && toggleContent.classList.contains('collapsed')) { toggleContent.classList.remove('collapsed'); const header = toggleContent.previousElementSibling; if (header) header.classList.add('active'); } } } const btn = $('focusModeBtn'); if (btn) btn.style.display = 'inline-block'; // Mostra o botão mágico } }
